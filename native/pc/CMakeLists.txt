cmake_minimum_required(VERSION 3.20)
project(jellyxr_mpv LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(GODOT_CPP_PATH "" CACHE PATH "Path to godot-cpp godot-4.5-stable")
set(MPV_ROOT "" CACHE PATH "Path to the libmpv development archive")
set(FFMPEG_ROOT "" CACHE PATH "Path to the FFmpeg shared development archive on Windows")
if(NOT GODOT_CPP_PATH)
    message(FATAL_ERROR "Set GODOT_CPP_PATH. See tools/build-pc-video.ps1 or tools/build-pc-video.sh.")
endif()
set(GODOTCPP_TARGET template_release CACHE STRING "" FORCE)
set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/build_profile.json" CACHE FILEPATH "" FORCE)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    set(GODOTCPP_USE_STATIC_CPP OFF CACHE BOOL "Use the host C++ runtime for graphics driver compatibility" FORCE)
endif()
add_subdirectory("${GODOT_CPP_PATH}" godot-cpp)
if(MINGW)
    target_compile_options(godot-cpp PRIVATE -include cstdlib)
endif()
if(WIN32)
    if(NOT MPV_ROOT)
        message(FATAL_ERROR "Set MPV_ROOT to the Windows libmpv development archive.")
    endif()
    find_path(MPV_INCLUDE_DIR mpv/client.h PATHS "${MPV_ROOT}/include" REQUIRED NO_DEFAULT_PATH)
    find_library(MPV_LIBRARY NAMES mpv mpv.dll libmpv.dll PATHS "${MPV_ROOT}" "${MPV_ROOT}/lib" REQUIRED NO_DEFAULT_PATH)
    add_library(jellyxr_mpv_dependency UNKNOWN IMPORTED)
    set_target_properties(jellyxr_mpv_dependency PROPERTIES
        IMPORTED_LOCATION "${MPV_LIBRARY}"
        INTERFACE_INCLUDE_DIRECTORIES "${MPV_INCLUDE_DIR}")
    set(MPV_TARGET jellyxr_mpv_dependency)
    find_path(FFMPEG_INCLUDE_DIR libavcodec/avcodec.h PATHS "${FFMPEG_ROOT}/include" REQUIRED NO_DEFAULT_PATH)
    foreach(component avcodec avformat avutil swscale)
        find_library(FFMPEG_${component}_LIBRARY NAMES ${component} PATHS "${FFMPEG_ROOT}/lib" REQUIRED NO_DEFAULT_PATH)
        list(APPEND FFMPEG_LIBRARIES "${FFMPEG_${component}_LIBRARY}")
    endforeach()
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(MPV REQUIRED IMPORTED_TARGET mpv>=2.0)
    set(MPV_TARGET PkgConfig::MPV)
    pkg_check_modules(FFMPEG REQUIRED IMPORTED_TARGET libavcodec>=60 libavformat>=60 libavutil>=58 libswscale>=7)
    set(FFMPEG_LIBRARIES PkgConfig::FFMPEG)
else()
    message(FATAL_ERROR "The PC video plugin supports Windows and Linux.")
endif()
find_package(Threads REQUIRED)

add_library(jellyxr_mpv SHARED
    mpv_video.cpp render_worker.cpp render_gl.cpp render_gl_worker.cpp
    gl_functions.cpp register_types.cpp render_vulkan.cpp
    video_decoder.cpp video_decoder_frames.cpp video_decoder_packets.cpp video_decoder_planes.cpp video_plane.cpp video_sync.cpp)
target_link_libraries(jellyxr_mpv PRIVATE godot-cpp ${MPV_TARGET} ${FFMPEG_LIBRARIES} Threads::Threads)
if(WIN32)
    target_include_directories(jellyxr_mpv PRIVATE "${FFMPEG_INCLUDE_DIR}")
endif()
if(WIN32)
    target_sources(jellyxr_mpv PRIVATE shared_gl_context_windows.cpp)
    target_link_libraries(jellyxr_mpv PRIVATE opengl32)
else()
    target_sources(jellyxr_mpv PRIVATE shared_gl_context_linux.cpp)
    target_link_libraries(jellyxr_mpv PRIVATE ${CMAKE_DL_LIBS})
endif()
if(MINGW)
    target_link_options(jellyxr_mpv PRIVATE -static -static-libgcc -static-libstdc++)
endif()
set_target_properties(jellyxr_mpv PROPERTIES PREFIX "" OUTPUT_NAME "jellyxr_mpv")
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    get_target_property(JELLYXR_ARCH godot-cpp GODOTCPP_ARCH)
    if(NOT JELLYXR_ARCH MATCHES "^(x86_64|arm64)$")
        message(FATAL_ERROR "Unsupported Linux architecture: ${JELLYXR_ARCH}")
    endif()
    set_target_properties(jellyxr_mpv PROPERTIES OUTPUT_NAME "jellyxr_mpv.linux.${JELLYXR_ARCH}")
endif()
