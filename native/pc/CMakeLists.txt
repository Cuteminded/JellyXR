cmake_minimum_required(VERSION 3.20)
project(jellyframe_mpv LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(GODOT_CPP_PATH "" CACHE PATH "Path to godot-cpp godot-4.5-stable")
set(MPV_ROOT "" CACHE PATH "Path to the libmpv development archive")
if(NOT GODOT_CPP_PATH OR NOT MPV_ROOT)
    message(FATAL_ERROR "Set GODOT_CPP_PATH and MPV_ROOT. See tools/build-pc-video.ps1.")
endif()
set(GODOTCPP_TARGET template_release CACHE STRING "" FORCE)
set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/build_profile.json" CACHE FILEPATH "" FORCE)
add_subdirectory("${GODOT_CPP_PATH}" godot-cpp)
if(MINGW)
    target_compile_options(godot-cpp PRIVATE -include cstdlib)
endif()
find_path(MPV_INCLUDE_DIR mpv/client.h PATHS "${MPV_ROOT}/include" REQUIRED NO_DEFAULT_PATH)
find_library(MPV_LIBRARY NAMES mpv mpv.dll libmpv.dll PATHS "${MPV_ROOT}" "${MPV_ROOT}/lib" REQUIRED NO_DEFAULT_PATH)
find_package(Threads REQUIRED)

add_library(jellyframe_mpv SHARED mpv_video.cpp render_worker.cpp register_types.cpp)
target_include_directories(jellyframe_mpv PRIVATE "${MPV_INCLUDE_DIR}")
target_link_libraries(jellyframe_mpv PRIVATE godot-cpp "${MPV_LIBRARY}" Threads::Threads)
if(MINGW)
    target_link_options(jellyframe_mpv PRIVATE -static -static-libgcc -static-libstdc++)
endif()
set_target_properties(jellyframe_mpv PROPERTIES PREFIX "" OUTPUT_NAME "jellyframe_mpv")
